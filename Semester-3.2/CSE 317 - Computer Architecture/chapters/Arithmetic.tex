   $\begin{array}{r}
      \text{5} = \text{0101} \\[-0.5em]
      \text{-6} = \text{1010} \\[-0.5em]
      \text{6} = \text{0110} \\[2em]
   \end{array}$\\
   \begin{tabular}{l}
      \textbf{\textit{Operations}:} \\
      10 = Sub m \\[-0.25em]
      01 = Add m \\[-0.25em]
      00/11 = No operation
   \end{tabular}\\
   
   \vspace{10em}
   \textbf{\textit{Answer}:} \\
   $\therefore \text{5} \times \text{-6}
   = (\text{1111 0010})_\text{2}
   = (\text{-30})_\text{10}$\\
