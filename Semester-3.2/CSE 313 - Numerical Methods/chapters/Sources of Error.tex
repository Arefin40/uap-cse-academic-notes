\chapter{Sources of Error}

\section{Round-off Error}
Round-off error created due to approximate representation of numbers.

\begin{center}
   $\frac{\text{1}}{\text{3}} \approx \text{0.3333} \ldots$

   $\sqrt{2} = 1.4142 \ldots$
\end{center}


\section{Truncation error}
Truncation error caused by approximating a mathematical process by truncating an infinite series.

\begin{center}
   Truncation Error $=$ Actual Value $-$ Truncated Approximate Value
\end{center}


\subsubsection{Example}
The Maclaurin series for $e^{1.2}$ is given by the summation series
\quad $e^{1.2}=1+1.2+\frac{1.2^2}{2!}+\frac{1.2^3}{3!}+\ldots$

Find the truncation error if the first four terms of the series are used.

\textbf{Solution:}

$e^{1.2} = 1+1.2+\frac{1.2^2}{2!}+\frac{1.2^3}{3!} = 3.208$

$\begin{aligned}
   \therefore \text{Truncation Error} &= \text{Exact value} - \text{Truncated Approximate value} \\
   &= 3.3201 - 3.208 \\
   &= 0.1121
\end{aligned}$
