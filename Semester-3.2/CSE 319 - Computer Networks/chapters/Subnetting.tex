\begin{itemize}[parsep=0.5em, left=0pt]
   \item \textbf{Network address:} All network bits \& subnet bits as it is, \rgbR{host bits all 0}
   \item \textbf{Broadcast address:} All network bits \& subnet bits as it is, \rgbR{host bits all 1}
   \item \textbf{Subnet mask:} All network bits \& subnet bits 1, \rgbR{host bits all 0}
   \item \textbf{Number of subnet:} $2^X$; where $X$ is the number of subnet bits
   \item \textbf{Number of host:} $2^Y-2$; where $Y$ is the number of host bits
   \item \textbf{Block size:} 256 - Subnet mask
\end{itemize}

\section{Subnetting Example}
\textbf{\textit{Question:}} 
A software company has just bought an IP address of 190.168.0.0/22 and it
needs to make subnets for the following network topology:
Network-1: (120 hosts), \ Network-2: (55 hosts), \ Network-3: (400 hosts).

\textbf{Solution:}\\[0.5em]
{\setlength{\tabcolsep}{1pt}
\begin{tabular}{lll@{\hspace{1em}}l}
   IP address &:& 190.168.0.0/22 \ (Class B)\\
   
   IP range &:&
   10111110 . 10101000 . \rgbB{000000} \rgbR{00} . \rgbR{00000000}
   $\to$ 190.168.0.0\\
   && 10111110 . 10101000 . \rgbB{000000} \rgbR{11} . \rgbR{11111111}
   $\to$ 190.168.3.255 \\[0.75em]

   % Number of subnets &:&
   % $\text{2}^{\text{6}} = \text{64}$ \\[-0.25em]

   % Number of valid hosts &:&
   % $\text{2}^{\text{10}} - \text{2} = \text{1022}$
\end{tabular}}
 
\textbf{Network-3:}\\[0.5em]
{\setlength{\tabcolsep}{1pt}
\begin{tabular}{lll}
   Hosts required &:& 400 \\
   Nearest power of 2 &:& $\text{2}^\text{9} = \text{512}$
   \quad $\to$ 32$-$9$=$23\\
   Subnet address &:&
   10111110 . 10101000 . \rgbB{0000000}\rgbR{0} . \rgbR{00000000}
   \ $\leftarrow$ \ 190.168.0.0/23\\

   Subnet Mask &:&
   11111111 . 11111111 . 1111111\rgbR{0} . \rgbR{00000000} \quad
   = 255 . 255 . 254 . {0} \\
   
   Network address &:&
   10111110 . 10101000 . 0000000\rgbR{0} . \rgbR{00000000} \quad
   = 190 . 168 . 0 . {0} \\
   
   Broadcast address &:&
   10111110 . 10101000 . 0000000\rgbR{1} . \rgbR{11111111} \quad
   = 190 . 168 . 1 . {255} \\
\end{tabular}}
