\Question{Design a logic circuit that follows the following requirements:}
A 4-bit i binary number is represented as D,C,B,A where D is MSB and A is LSB. Design a logic circuit that will produce a high output whenever the binary number is greater than 0010 and less than 1010 or all inputs are low or all inputs are high.

\Solution
\begin{tabular}{lcl}
   \begin{adjustbox}{valign=t}
      \begin{tabularx}{7cm}{l|C|C|C|C|C|cl}\cline{2-6}

         & $\to$ & $\not{ABCD}$ \\\cline{2-6}

         & $\to$ & $ABC\not{D}$ \\\cline{2-6}
         8 & 1 & 0 & 0 & 0 & 1
         & $\to$ & $\not{ABC}D$ \\\cline{2-6}
         9 & 1 & 0 & 0 & 1 & 1

\Question{Design a logic circuit that follows the following requirements:}
Suppose A,B,C,D are inputs and Y is the output. Output will be high when A=B and C≠D.

\vspace{1em}
\Solution
\begin{tabular}{lcl}
   \begin{adjustbox}{valign=t}
      \begin{tabularx}{7cm}{l|C|C|C|C|C|cl}\cline{2-6}
         & A & B & C & D & Y \\\cline{2-6}
         0 & 0 & 0 & 0 & 0 & 0 \\\cline{2-6}
         1 & 0 & 0 & 0 & 1 & 1
         & $\to$ & $\not{ABC}D$ \\\cline{2-6}
         2 & 0 & 0 & 1 & 0 & 1 
         & $\to$ & $\not{AB}C\not{D}$ \\\cline{2-6}
         3 & 0 & 0 & 1 & 1 & 0 \\\cline{2-6}
         \\\cline{2-6}
         4 & 0 & 1 & 0 & 0 & 0 \\\cline{2-6}
         5 & 0 & 1 & 0 & 1 & 0 \\\cline{2-6}
         6 & 0 & 1 & 1 & 0 & 0 \\\cline{2-6}
         7 & 0 & 1 & 1 & 1 & 0 \\\cline{2-6}
         8 & 1 & 0 & 0 & 0 & 0 \\\cline{2-6}
         9 & 1 & 0 & 0 & 1 & 0 \\\cline{2-6}
         10 & 1 & 0 & 1 & 0 & 0 \\\cline{2-6}
         11 & 1 & 0 & 1 & 1 & 0 \\\cline{2-6}
         \\\cline{2-6}
         12 & 1 & 1 & 0 & 0 & 0 \\\cline{2-6}
         13 & 1 & 1 & 0 & 1 & 1
         & $\to$ & $AB\not{C}D$ \\\cline{2-6}
         14 & 1 & 1 & 1 & 0 & 1
         & $\to$ & $ABC\not{D}$ \\\cline{2-6}
         15 & 1 & 1 & 1 & 1 & 0 \\\cline{2-6}

            % \NumberGridO;

            \implicant{1}{1}
            \implicant{2}{2}
            \implicant{13}{13}
            \implicant{14}{14}
         \end{Karnaugh}\\
         $Y = \not{ABC}D + \not{AB}C\not{D} + AB\not{C}D + ABC\not{D}$
      \end{tabular}
   \end{adjustbox}
\end{tabular}
