\Question{Describe universality of NAND Gate.}
We can design any gate using only NAND gate, therefore NAND gate is an universal gate.

\begin{tabularx}{0.98\textwidth}{XX}
   \textbf{NOT gate:}  & \textbf{AND gate:} \\[1em]
   \begin{circuitikz}[y=-1cm]
      \draw (3,1) node[nand port, anchor=out] (nand){}
      (nand.in 1) -- ++(-0.2,0) coordinate (X) --
      ++(-0.5,0) node[left]{$A$}
      (nand.in 2) -| (X)
      (nand.out) -- ++(0.4,0) node[right]{$\not{A\cdot A} = \not{A}$};
   \end{circuitikz} & 
   \begin{circuitikz}[y=-1cm]
      \draw
      (3,1) node[nand port, anchor=out] (nand1){}
      (nand1.in 1) -- ++(-0.25,0) node[left]{$A$}
      (nand1.in 2) -- ++(-0.25,0) node[left]{$B$}
   
      (nand1.out) -- ++(0.25,0)
      (5.25,1) node[nand port, anchor=out] (nand2){}
      (nand2.in 2) -- ++(-0.25,0) |- (nand2.in 1)
      (nand2.out) -- ++(0.25,0) node[right]{$\not{\not{AB}} = AB$};
   \end{circuitikz} \\[2em]

   \textbf{OR gate:}  & \textbf{XOR gate:}\\[1em]
   \begin{circuitikz}[y=-1cm]
      \draw
      (3,1) node[nand port, anchor=out] (nand1){}
      (nand1.in 1) -- ++(-0.75,0) node[left]{$A$}
      (nand1.in 2) -- ++(-0.25,0) |- (nand1.in 1)
      
      (3,2.5) node[nand port, anchor=out] (nand2){}
      (nand2.in 1) -- ++(-0.75,0) node[left]{$B$}
      (nand2.in 2) -- ++(-0.25,0) |- (nand2.in 1)
      
      (5,1.75) node[nand port, anchor=out] (nand3){}
      (nand3.in 1) |- (nand1.out)
      (nand3.in 2) |- (nand2.out)
      
      (nand3.out) -- ++(0.25,0)
      node[right]{$A+B$};
   \end{circuitikz} & 
   \begin{circuitikz}[y=-1cm]
      \draw
      (1,1.75) node[nand port, anchor=out] (nand0){}
      (3,1) node[nand port, anchor=out] (nand1){}
      (3,2.5) node[nand port, anchor=out] (nand2){}
      (5,1.75) node[nand port, anchor=out] (nand3){}
   
      (nand0.out) -| (nand1.in 2)
      (nand0.out) -| (nand2.in 1)
   
      (nand3.in 1) |- (nand1.out)
      (nand3.in 2) |- (nand2.out)
      
      (nand1.in 1) -| (nand0.in 1)
      (nand2.in 2) -| (nand0.in 2)
      
      (nand1.in 1) to[short,-*] ++(-2,0) -- 
      ++(-0.5,0) node[left]{$A$}
      (nand2.in 2) to[short,-*] ++(-2,0) -- 
      ++(-0.5,0) node[left]{$B$}
      
      (nand3.out) -- ++(0.25,0) node[right]{$A \oplus B$};
   \end{circuitikz} \\
   $\not{\not{A}\not{B}} = \not{\not{A}}+\not{\not{B}} = A+B$
   & $\begin{aligned}
   &\not{\not{A\not{AB}}\cdot \not{B\not{AB}}}
   = A\not{AB} + B\not{AB}\\
   & = A(\not{A}+\not{B})+B(\not{A}+\not{B})\\
   & = (\not{A}+\not{B})(A+B) = A \oplus B
   \end{aligned}$
\end{tabularx}

\Question{Describe universality of NOR Gate.}
We can design any gate using only NOR gate, therefore NOR gate is an universal gate.

\begin{tabularx}{0.98\textwidth}{XX}
   \textbf{NOT gate:}  & \textbf{OR gate:}\\[1em]
   \begin{circuitikz}[y=-1cm]
      \draw (3,1) node[nor port, anchor=out] (nor){}
      (nor.in 1) -- ++(-0.2,0) coordinate (X) --
      ++(-0.5,0) node[left]{$A$}
      (nor.in 2) -| (X)
      (nor.out) -- ++(0.4,0)
      node[right]{$\not{A+A}=\not{A}$};
   \end{circuitikz}
   & 
   \begin{circuitikz}[y=-1cm]
      \draw
      (3,1) node[nor port, anchor=out] (nor1){}
      (nor1.in 1) -- ++(-0.25,0) node[left]{$A$}
      (nor1.in 2) -- ++(-0.25,0) node[left]{$B$}
   
      (nor1.out) -- ++(0.25,0)
      (5.25,1) node[nor port, anchor=out] (nor2){}
      (nor2.in 2) -- ++(-0.25,0) |- (nor2.in 1)
      (nor2.out) -- ++(0.25,0) node[right]{$\not{\not{A+B}} = A+B$};
   \end{circuitikz}\\[2em]

   \textbf{AND gate:}  & \textbf{XOR gate:}\\[1em]
   \begin{adjustbox}{valign=t}
      \begin{tabular}{l}
         \begin{circuitikz}[y=-1cm]
            \draw
            (3,1) node[nor port, anchor=out] (nor1){}
            (nor1.in 1) -- ++(-0.75,0) node[left]{$A$}
            (nor1.in 2) -- ++(-0.25,0) |- (nor1.in 1)
            
            (3,2.5) node[nor port, anchor=out] (nor2){}
            (nor2.in 1) -- ++(-0.75,0) node[left]{$B$}
            (nor2.in 2) -- ++(-0.25,0) |- (nor2.in 1)
            
            (5,1.75) node[nor port, anchor=out] (nor3){}
            (nor3.in 1) |- (nor1.out)
            (nor3.in 2) |- (nor2.out)
            
            (nor3.out) -- ++(0.25,0) node[right]{$AB$};
         \end{circuitikz}\\[1em]
         $\not{\not{A}+\not{B}} = \not{\not{A}}\not{\not{B}} = AB$
      \end{tabular}
   \end{adjustbox}
   & 
   \begin{adjustbox}{valign=t}
      \begin{circuitikz}[y=-1cm]
         \draw
         (3,1) node[nor port, anchor=out] (nor1){}
         (1,2.5) node[nor port, anchor=out] (nor2){}
         (1,4) node[nor port, anchor=out] (nor3){}
         (3,3.25) node[nor port, anchor=out] (nor4){}
         (5,2) node[nor port, anchor=out] (nor5){}
   
         (nor2.in 2) |- (nor1.in 1)
         (nor3.in 1) -- ++(-0.5,0)
         (nor3.in 2) -- ++(-0.5,0) |- (nor1.in 2)
   
         (nor4.in 1) |- (nor2.out)
         (nor4.in 2) |- (nor3.out)
   
         (nor1.out) -| (nor5.in 1)
         (nor4.out) -| (nor5.in 2)
         
         (nor1.in 1) to[short,-*] ++(-2,0)
         -- ++(-1,0) node[left]{$A$}
         (nor1.in 2) to[short,-*] ++(-2.5,0)
         -- ++(-0.5,0) node[left]{$B$}
   
         (nor1.out) node[above right]{$\not{A+B}$}
         (nor2.out) node[above]{$\not{A}$}
         (nor3.out) node[below]{$\not{B}$}
         (nor4.out) node[below right]{$AB$}
   
         (nor5.out) -- ++(0.25,0) node[right]{$A \oplus B$};
      \end{circuitikz}
   \end{adjustbox} \\
   & \vspace{0em}
   $\not{(\not{A+B})+(AB)}
   = (\not{\not{A+B}}) \cdot (\not{A}+\not{B})
   = A \oplus B$
\end{tabularx}
