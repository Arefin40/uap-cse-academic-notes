\newcommand{\Xmark}[2][G]{%
    \ifx#1success\relax
        \xdef\BGcolor{rgbG}
        \xdef\FGcolor{rgbGF}
    \else\ifx#1G\relax
        \xdef\BGcolor{rgbG}
        \xdef\FGcolor{rgbGF}
    \else\ifx#1danger\relax
        \xdef\BGcolor{rgbR}
        \xdef\FGcolor{rgbRF}
    \else\ifx#1R\relax
        \xdef\BGcolor{rgbR}
        \xdef\FGcolor{rgbRF}
    \else\ifx#1info\relax
        \xdef\BGcolor{rgbB}
        \xdef\FGcolor{rgbBF}
    \else\ifx#1B\relax
        \xdef\BGcolor{rgbB}
        \xdef\FGcolor{rgbBF}
    \else\ifx#1warning\relax
        \xdef\BGcolor{rgbY}
        \xdef\FGcolor{rgbYF}
    \else\ifx#1Y\relax
        \xdef\BGcolor{rgbY}
        \xdef\FGcolor{rgbYF}
    \fi\fi\fi\fi\fi\fi\fi\fi
    \raisebox{0.5\height}{\tikz[baseline]{\filldraw[\BGcolor] (0,0) circle(6pt); \node[\FGcolor] at (0,0) {\small\textbf{#2}};}}%
}

% Question Container
\makeatletter
\newcommand{\IfEmptyTF}[3]{%
  \edef\temp{#1}%
  \ifx\temp\@empty
    #2% true (empty)
  \else
    #3% false (non-empty)
  \fi
}
\makeatother
\NewDocumentEnvironment{Question}{O{} O{} +b}{%
  \noindent
  \IfEmptyTF{#1}{% marks empty
    \IfEmptyTF{#2}{% CO also empty
      % --- both empty → only body ---
      #3%
    }{%
      % --- only CO ---
      \begin{minipage}[t]{0.9\textwidth}#3\end{minipage}%
      \hfill
      \begin{minipage}[t]{0.1\textwidth}\raggedleft #2\end{minipage}\par
    }%
  }{%
    % --- marks present (maybe CO too) ---
    \begin{minipage}[t]{0.9\textwidth}#3\end{minipage}%
    \hfill
    \begin{minipage}[t]{0.1\textwidth}\raggedleft
      [#1]%
      \IfEmptyTF{#2}{}{\\#2}%
    \end{minipage}\par
  }%
}{}
