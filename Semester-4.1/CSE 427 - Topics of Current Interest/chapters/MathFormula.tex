\topic{Tower of Hanoi}
\subtopic{Simple Tower of Hanoi}
\begin{tabular}{ll}
   Base case: & $T_1 = 1$ \\
   Recursive formula: & $T_n = 2T_{n-1} + 1$ \\
   Closed form: & $T_n = 2^n - 1$
\end{tabular}

\subtopic{Double Tower of Hanoi}
\begin{tabular}{ll}
   Recursive formula: & $T_n = 2T_{n-1} + 2$ \\
   Closed form: & $T_n = 2(2^n - 1)$
\end{tabular}

\subtopic{Triple Tower of Hanoi}
\begin{tabular}{ll}
   Recursive formula: & $T_n = 2T_{n-1} + 3$ \\
   Closed form: & $T_n = 3(2^n - 1)$
\end{tabular}

\topic{Josephus Problem}
\begin{tabular}{ll}
   For even &: $J(2n) = 2J(n)-1$\\
   For odd &: $J(2n+1) = 2J(n) - 1$
\end{tabular}

\subtopic{Close Form}
Survival Position, $n = 2l+1$\\[0.5em]
Participants, $n = 2^m + l$\\[0.5em]
$J(n) \to n^{th}  \quad \Rightarrow J(2^m + l) = 2l+1$

\subtopic{Binary Property}
$\begin{aligned}
   & n = (b_m b_{m-1} b_{m-2} \ldots\ldots b_1 b_0)_2 \\
   & l = (0 b_{m-1} b_{m-2} \ldots\ldots b_1 b_0)_2 \\
   & 2l = (b_{m-1} b_{m-2} \ldots\ldots b_1 b_0 0)_2 \\
   & 2l+1 = (b_{m-1} b_{m-2} \ldots\ldots b_1 b_0 1)_2
\end{aligned}$

\topic{Binomial Distribution}
Number of favorable outcomes: $C(n, k) = \frac{n!}{k!(n-k)!}$

Probability of each outcome: $p^k \times (1-p)^{n-k}$

Probability: $P(X=k) = C(n, k) \times$ Probability of each outcome

Where\\
$n$ = number of trials \\
$k$ = number of successful trials \\
$p$ = probability of success on an individual trial




% ==========================================
\topic{Poisson Distribution}
Probability: $P(X=k) = \frac{\lambda^k e^{-\lambda}}{k!}$

Where\\
$k$ = number of occurrences of an event \\
$\lambda$ = average number of occurrences in a given interval \\




\topic{Probability}
Union/OR: $P(A \cup B) = P(A) + P(B) - P(A\cap B)$

Bayes' Theorem: $P(A|B) = \frac{P(B|A) \cdot P(A)}{P(B)}$

Total Probability: $P(B) = P(B|A) \cdot P(A) + P(B|A^c) \cdot P(A^c)$

Conditional Probability: $P(A|B) = \frac{P(A \cap B)}{P(B)}$
