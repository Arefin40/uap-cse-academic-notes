\newcommand{\drawX}[3][]{%
   \def\x{0.35}\def\y{0.65}
   \pgfmathsetmacro{\flipy}{2-#3}
   \ifthenelse{\equal{#1}{filled}}{%
      \fill[start!20] (#2+0.5,\flipy+0.5) circle [radius=0.325];
   }{}%
   \draw[thick,rgbGF] (#2+\x,\flipy+\x) -- (#2+\y,\flipy+\y);
   \draw[thick,rgbGF] (#2+\y,\flipy+\x) -- (#2+\x,\flipy+\y);
}
\newcommand{\drawO}[3][]{%
   \def\r{0.2}
   \pgfmathsetmacro{\flipy}{2-#3}
   \ifthenelse{\equal{#1}{filled}}{%
      \fill[start!20] (#2+0.5,\flipy+0.5) circle [radius=0.325];
   }{}%
   \draw[thick,rgbRF] (#2+0.5,\flipy+0.5) circle [radius=\r];
}
\newcommand{\cell}[4][]{%
   \pgfmathsetmacro{\flipy}{2-#3}
   \ifthenelse{\equal{#1}{filled}}{%
      \fill[start!30] (#2+0.5,\flipy+0.5) circle [radius=0.325];
   }{}%
   \node at (#2+0.5,\flipy+0.5) {#4};
}
