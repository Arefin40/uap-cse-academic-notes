\subsubsection{Simple Tower of Hanoi}
\begin{tabular}{ll}
   Base case: & $T_1 = 1$ \\
   Recursive formula: & $T_n = 2T_{n-1} + 1$ \\
   Closed form: & $T_n = 2^n - 1$
\end{tabular}

\subsubsection{Double Tower of Hanoi}
\begin{tabular}{ll}
   Recursive formula: & $T_n = 2T_{n-1} + 2$ \\
   Closed form: & $T_n = 2(2^n - 1)$
\end{tabular}

\subsubsection{Triple Tower of Hanoi}
\begin{tabular}{ll}
   Recursive formula: & $T_n = 2T_{n-1} + 3$ \\
   Closed form: & $T_n = 3(2^n - 1)$
\end{tabular}

\columnbreak

\subsubsection{Lines in the plane}
\begin{tabular}{ll}
   Base case: & $L_0 = 1$ \\
   Recursive formula: & $L_n = L_{n-1} + n$ \\
   Closed form: & $L_n = \frac{n(n+1)}{2}+1$
\end{tabular}


\subsubsection{Zig}
$\begin{aligned}
\text{Regions:} \quad Z_n &= L_{2n} - 2n\\
       &= \frac{2n(2n+1)}{2} + 1 - 2n \\
      %  &= 2n^2 - n + 1
\end{aligned}$

$A=2n \quad \rgbR{:} \quad B = L_{2n} \quad \rgbR{:} \quad B - 2n \quad \rgbR{:} \ 2n$\\[1em]
$A=2X \quad \rgbR{:} \quad B = \frac{A(A+1)}{2}+1 \quad \rgbR{:} \quad B - 2X \quad \rgbR{:} \quad 2X$

\subsubsection{Zig Zag}
$\begin{aligned}

       &= \frac{3n(3n+1)}{2} + 1 - 5n \\
      %  &= \frac{9n^2 - 7n + 2}{2}
